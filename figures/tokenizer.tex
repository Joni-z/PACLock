% PACLock hybrid tokenizer -- architecture schematic for talks. v2 layout:
% two straight lanes (raw above, PAC below) converging symmetrically on the
% hybrid grid; every lane box on one baseline, notes kept outside the lanes.
% Compile: pdflatex tokenizer.tex   (standalone; standard TikZ + amsmath only)
\documentclass[tikz,border=10pt]{standalone}
\usepackage{amsmath,amssymb}
\usetikzlibrary{positioning,arrows.meta,calc,decorations.pathreplacing}

\definecolor{rawc}{RGB}{56,108,176}     % raw path
\definecolor{pacc}{RGB}{217,120,44}     % PAC interaction path
\definecolor{annot}{RGB}{105,105,105}

\tikzset{
  box/.style={draw=black!55, rounded corners=2pt, fill=black!4,
              align=center, font=\small, inner sep=5pt, minimum height=16mm},
  rawbox/.style={box, draw=rawc, fill=rawc!9},
  pacbox/.style={box, draw=pacc, fill=pacc!9},
  flow/.style={-{Stealth[length=2.4mm]}, thick, black!60},
  rawflow/.style={flow, rawc!85!black},
  pacflow/.style={flow, pacc!85!black},
  note/.style={font=\scriptsize\itshape, text=annot, align=center},
  lbl/.style={font=\scriptsize, align=center},
}

\begin{document}
\begin{tikzpicture}

% ============================================================ lane geometry
\def\yraw{20mm}     % raw lane baseline
\def\ypac{-20mm}    % PAC lane baseline

% ------------------------------------------------------------ shared input
\node[box, minimum height=13mm] (input) at (0,0)
  {raw EEG\\ $x\in\mathbb{R}^{C\times T}$};
\node[box, minimum height=13mm, right=9mm of input] (sinc)
  {learnable sinc\\ filterbank\\ {\scriptsize $n_b$ band-pass}};
\draw[flow] (input) -- (sinc);

% fan-out point
\coordinate (split) at ($(sinc.east)+(5mm,0)$);
\draw[flow] (sinc.east) -- (split);

% ============================================================ RAW lane
\node[rawbox, text width=34mm] (rawtok) at (78mm,\yraw)
  {\textbf{raw patch tokenizer}\\[0.5mm]
   $r_j=\mathrm{Conv1d}(x_j)$\\[0.5mm]
   {\scriptsize one token per (band, patch)}};
\draw[rawflow] (split) |- node[pos=0.72, above, lbl, rawc]
  {band signals $x_j$} (rawtok.west);
\node[note, above=2mm of rawtok, text width=70mm]
  {band $j$ keeps its \textbf{own} phase and waveform\\
   $\rightarrow$ within-band morphology (seizure, motor imagery)};

% ============================================================ PAC lane
\node[pacbox, text width=22mm] (hilb) at (58mm,\ypac)
  {Hilbert\\[0.5mm] $e^{i\varphi_j},\;A_j$};
\draw[pacflow] (split) |- (hilb.west);

\node[pacbox, text width=30mm, right=6mm of hilb] (toks)
  {tokenizers\\[0.5mm]
   $p_j=L\big(e^{i\varphi_j}\big)$\\[0.5mm]
   $a_j=L'\big(\log(1{+}A_j)\big)$};
\draw[pacflow] (hilb) -- (toks);

\node[pacbox, text width=40mm, right=6mm of toks] (align)
  {\textbf{coupling + alignment}\\[0.5mm]
   $Z_{ij}=\big\langle(A_j{-}\bar A_j)\,e^{i\varphi_i}\big\rangle_{\text{patch}}$\\[0.5mm]
   $\tilde p_j=\textstyle\sum_{i<j}\alpha_{ij}\,e^{-i\angle Z_{ij}}\,p_i$};
\draw[pacflow] (toks) -- (align);

\node[pacbox, text width=30mm, right=6mm of align] (rot)
  {\textbf{rotation}\\[0.5mm]
   $h_j=a_j\odot \tilde p_j/|\tilde p_j|$\\[0.5mm]
   {\scriptsize $|h_j|=|a_j|$}};
\draw[pacflow] (align) -- (rot);

\node[circle, draw=pacc, fill=white, inner sep=1.4pt, font=\small,
      right=6mm of rot] (gate) {$\times\alpha_j$};
\draw[pacflow] (rot) -- (gate);

\node[note, below=2mm of align.south east, anchor=north, text width=70mm]
  {band $j$'s phase is replaced by a coupling-weighted mixture of lower
   bands' phases (gauge-invariant)\\
   $\rightarrow$ cross-band structure (epileptiform events);\quad
   $\times\alpha_j$: learned per-band gate, init $1$};

% ============================================================ hybrid grid
% 8 schematic rows (4 raw + 4 interaction), 6 columns, centred on y=0.
% Placed to the right of BOTH lanes so no lane box sits under it; the two
% paths converge on its top and bottom edges (raw fills the upper rows,
% interaction the lower), which keeps left and right edges free for the
% row braces and the encoder arrow.
\coordinate (g0) at (214mm,13mm);
\foreach \r in {0,1,2,3} {
  \foreach \c in {0,1,2,3,4,5} {
    \fill[rawc!25, draw=rawc!60, line width=0.3pt]
      ($(g0)+(\c*3.4mm,-\r*3.4mm)$) rectangle ++(3.1mm,-3.1mm);
  }
}
\foreach \r in {4,5,6,7} {
  \foreach \c in {0,1,2,3,4,5} {
    \fill[pacc!25, draw=pacc!60, line width=0.3pt]
      ($(g0)+(\c*3.4mm,-\r*3.4mm)$) rectangle ++(3.1mm,-3.1mm);
  }
}
\coordinate (gtopC) at ($(g0)+(10.2mm, 0.2mm)$);   % top edge, centre column
\coordinate (gbotC) at ($(g0)+(10.2mm,-26.9mm)$);  % bottom edge, centre column
\coordinate (gR)    at ($(g0)+(20.9mm,-13.4mm)$);  % right edge, vertical centre

\node[lbl, anchor=south east] at ($(g0)+(-4mm,0.5mm)$)
  {\textbf{hybrid token grid} $(C\times 2n_b\times P)$};
\draw[decorate, decoration={brace, amplitude=2.2mm, mirror}, rawc, thick]
  ($(g0)+(-1.8mm,0)$) -- ($(g0)+(-1.8mm,-13.2mm)$)
  node[midway, left=2.8mm, lbl, rawc, align=right] {raw\\ rows};
\draw[decorate, decoration={brace, amplitude=2.2mm, mirror}, pacc, thick]
  ($(g0)+(-1.8mm,-13.6mm)$) -- ($(g0)+(-1.8mm,-26.8mm)$)
  node[midway, left=2.8mm, lbl, pacc, align=right] {inter-\\ action\\ rows};

% convergence: raw enters through the TOP edge, interaction through the BOTTOM
\draw[rawflow] (rawtok.east) -| (gtopC);
\draw[pacflow] (gate.east)   -| (gbotC);

% ============================================================ encoder + head
\node[box, text width=27mm, minimum height=26mm, anchor=west]
  (enc) at ($(gR)+(9mm,0)$)
  {\textbf{tri-axial encoder}\\[1mm]
   attention over\\ time $\cdot$ electrodes $\cdot$\\ frequency rows\\[1mm]
   {\scriptsize chooses per task\\ which rows to read}};
\draw[flow] (gR) -- (enc.west);
\node[box, minimum height=12mm, right=6mm of enc] (head) {readout\\ head};
\draw[flow] (enc) -- (head);

% ============================================================ caption line
\node[note, text width=170mm, anchor=north west] at (12mm,-48mm)
  {Nothing is replaced: raw rows keep every band's own phase, interaction
   rows carry the measured cross-band coupling, and frequency attention plus
   the learned gate decide how much of each a task uses. The constitutive
   PAC tokenizer is the special case with the raw rows deleted; the raw
   tokenizer is the special case with the interaction rows deleted.};

\end{tikzpicture}
\end{document}
