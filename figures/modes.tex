\newcommand{\gridblock}[4]{%
  % #1 = origin coordinate name, #2 = rows, #3 = fill color, #4 = row offset
  \foreach \r in {1,...,#2} {
    \foreach \c in {0,1,2,3,4} {
      \fill[#3!25, draw=#3!60, line width=0.3pt]
        ($(#1)+(\c*3.2mm, -(\r-1)*3.2mm - #4)$) rectangle ++(3.0mm,-3.0mm);
    }
  }
}
